# SAIGE Showcase 01: a scripted, real-renderer demonstration (docs/showcase/01-courtyard.md).
# The course library needs only Core and the glTF/JSON parser; the renderer app needs the Vulkan backend.
if(TARGET saige::core AND TARGET saige::gfx_assets)
    add_library(saige_showcase_course STATIC 01-courtyard/src/course.cpp)
    target_include_directories(saige_showcase_course PUBLIC 01-courtyard/include)
    target_link_libraries(saige_showcase_course PUBLIC saige::core PRIVATE saige::gfx_assets)
    saige_gfx_configure_target(saige_showcase_course)

    if(BUILD_TESTING)
        add_executable(showcase_course_test 01-courtyard/tests/course_test.cpp)
        target_link_libraries(showcase_course_test PRIVATE saige_showcase_course)
        target_compile_definitions(showcase_course_test PRIVATE SAIGE_SHOWCASE_ASSETS="${CMAKE_CURRENT_SOURCE_DIR}/01-courtyard/assets")
        saige_gfx_configure_target(showcase_course_test)
        add_test(NAME showcase_course COMMAND showcase_course_test)
    endif()

    if(TARGET saige::rhi_vulkan AND TARGET saige::gfx_scene AND TARGET saige::render)
        add_executable(saige_showcase 01-courtyard/src/main.cpp)
        target_link_libraries(saige_showcase PRIVATE saige_showcase_course saige::rhi_vulkan saige::gfx_scene saige::gfx_assets saige::render)
        target_compile_definitions(saige_showcase PRIVATE SAIGE_SHOWCASE_DEFAULT_ASSETS="${CMAKE_CURRENT_SOURCE_DIR}/01-courtyard/assets")
        saige_gfx_configure_target(saige_showcase)
        if(BUILD_TESTING)
            # A short end-to-end run on the real renderer. Exit code 77 = no Vulkan 1.3 device: reported as skipped.
            add_test(NAME showcase_smoke COMMAND saige_showcase --out ${CMAKE_CURRENT_BINARY_DIR}/smoke --width 320 --height 180
                     --stills 1 --sequence-seconds 1 --validate --self-check)
            set_tests_properties(showcase_smoke PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 300)
        endif()
    endif()
endif()

# SAIGE Showcase 04 "A World Alive" (docs/showcase/04-world-alive.md): a coastal district and the living world on it. The district is pure data from a seed; the world joins it to the engine's
# simulation (world_sim, crowd, traffic, multi_floor_nav), physics (CapsuleCharacter, Vehicle, DynamicsWorld) and audio. Everything here builds without a GPU; the renderer app is a separate target.
if(TARGET saige::core)
    add_library(saige_alive_district STATIC
        04-world-alive/src/district.cpp 04-world-alive/src/district_interiors.cpp 04-world-alive/src/district_network.cpp 04-world-alive/src/district_validate.cpp 04-world-alive/src/world.cpp 04-world-alive/src/traffic_driver.cpp 04-world-alive/src/scout.cpp)
    target_include_directories(saige_alive_district PUBLIC 04-world-alive/src)
    target_link_libraries(saige_alive_district PUBLIC saige::core)
    if(TARGET saige::rhi_vulkan AND TARGET saige::gfx_scene AND TARGET saige::gfx_assets AND TARGET saige::render AND TARGET saige::animation)
        add_executable(saige_alive 04-world-alive/render/main.cpp 04-world-alive/render/gfx.cpp 04-world-alive/render/city.cpp 04-world-alive/render/actors.cpp 04-world-alive/render/director.cpp)
        target_include_directories(saige_alive PRIVATE 04-world-alive/src 04-world-alive/render)
        target_link_libraries(saige_alive PRIVATE saige_alive_district saige::animation saige::rhi_vulkan saige::gfx_scene saige::gfx_assets saige::render)
        saige_gfx_configure_target(saige_alive)
    endif()
    add_executable(alive_scout 04-world-alive/tools/scout_main.cpp)
    target_link_libraries(alive_scout PRIVATE saige_alive_district)
    add_executable(alive_map_dump 04-world-alive/tools/map_dump.cpp)
    target_link_libraries(alive_map_dump PRIVATE saige_alive_district)
    add_executable(alive_validate 04-world-alive/tools/validate_main.cpp)
    target_link_libraries(alive_validate PRIVATE saige_alive_district)
    if(BUILD_TESTING)
        add_executable(alive_district_test 04-world-alive/tests/district_test.cpp)
        target_link_libraries(alive_district_test PRIVATE saige_alive_district)
        add_test(NAME alive_district COMMAND alive_district_test)
        set_tests_properties(alive_district PROPERTIES TIMEOUT 120)
        add_executable(alive_validate_test 04-world-alive/tests/validate_test.cpp)
        target_link_libraries(alive_validate_test PRIVATE saige_alive_district)
        add_test(NAME alive_validate COMMAND alive_validate_test)
        set_tests_properties(alive_validate PROPERTIES TIMEOUT 120)
        # The chair model's front axis against the convention the generator and the SEAT_FACING rule assume (exit 77: the asset is not restored; a failure with SAIGE_REQUIRE_VISUAL=1).
        if(Python3_FOUND)
            add_test(NAME alive_prop_orientation COMMAND ${Python3_EXECUTABLE} -I ${CMAKE_CURRENT_SOURCE_DIR}/04-world-alive/tools/check_prop_orientation.py)
            set_tests_properties(alive_prop_orientation PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77)
        endif()
        # Visual regression: renders real film frames twice with one controlled difference (tools/visual_regression.py). Skipped (77) without the renderer or the texture cache. The fixture run
        # (--fixtures, generated stand-ins, needs no download) is a required step of the Graphics workflow rather than a ctest, because it needs a Vulkan device (the Windows runners have none).
        if(TARGET saige_alive)
            find_package(Python3 COMPONENTS Interpreter QUIET)
            if(Python3_Interpreter_FOUND)
                add_test(NAME alive_visual COMMAND ${Python3_EXECUTABLE} -I ${CMAKE_CURRENT_SOURCE_DIR}/04-world-alive/tools/visual_regression.py --app $<TARGET_FILE:saige_alive> --scout-tool $<TARGET_FILE:alive_scout>)
                set_tests_properties(alive_visual PROPERTIES TIMEOUT 1500 SKIP_RETURN_CODE 77)
            endif()
        endif()
        # The visual runner's own failure modes (corrupt frames, a sweep that fails, the wrong frame count, bad arguments): needs no renderer and no assets.
        find_package(Python3 COMPONENTS Interpreter QUIET)
        if(Python3_Interpreter_FOUND)
            add_test(NAME alive_visual_selftest COMMAND ${Python3_EXECUTABLE} -I ${CMAKE_CURRENT_SOURCE_DIR}/04-world-alive/tools/visual_regression_selftest.py)
            set_tests_properties(alive_visual_selftest PROPERTIES TIMEOUT 120)
        endif()
        # The scout file format and the director's use of it. The director is compiled in directly (it needs the renderer's headers, not a GPU).
        if(TARGET saige_alive)
            add_executable(alive_scout_test 04-world-alive/tests/scout_test.cpp 04-world-alive/render/director.cpp)
            target_include_directories(alive_scout_test PRIVATE 04-world-alive/render)
            target_link_libraries(alive_scout_test PRIVATE saige_alive_district saige::gfx_assets saige::render)
            add_test(NAME alive_scout COMMAND alive_scout_test)
            set_tests_properties(alive_scout PROPERTIES TIMEOUT 300)
            # The director's camera over whole films (no renderer, no textures): the full matrix of seeds and reports, and two mutants that must fail (the film's original behaviour, and a displacement budget too small for the worlds).
            add_executable(alive_camera_test 04-world-alive/tests/camera_test.cpp 04-world-alive/render/director.cpp)
            target_include_directories(alive_camera_test PRIVATE 04-world-alive/render)
            target_link_libraries(alive_camera_test PRIVATE saige_alive_district saige::gfx_assets saige::render)
            add_test(NAME alive_camera COMMAND alive_camera_test)   # the default matrix (4 cases); `alive_camera_test --all` walks all 16 (8 seeds x 2 reports)
            set_tests_properties(alive_camera PROPERTIES TIMEOUT 1800)
            add_test(NAME alive_camera_mutant_film_behaviour COMMAND alive_camera_test 3:real)
            set_tests_properties(alive_camera_mutant_film_behaviour PROPERTIES TIMEOUT 900 WILL_FAIL TRUE ENVIRONMENT "SAIGE_ALIVE_NO_CANOPY_CLEAR=1")
            add_test(NAME alive_camera_mutant_small_budget COMMAND alive_camera_test 2:real)
            set_tests_properties(alive_camera_mutant_small_budget PROPERTIES TIMEOUT 900 WILL_FAIL TRUE ENVIRONMENT "SAIGE_ALIVE_EYE_BUDGET=1.0")
        endif()
        add_executable(alive_world_test 04-world-alive/tests/world_test.cpp)
        target_link_libraries(alive_world_test PRIVATE saige_alive_district)
        add_test(NAME alive_world COMMAND alive_world_test)
        set_tests_properties(alive_world PROPERTIES TIMEOUT 900)
        add_executable(alive_traffic_test 04-world-alive/tests/traffic_test.cpp)
        target_link_libraries(alive_traffic_test PRIVATE saige_alive_district)
        add_test(NAME alive_traffic COMMAND alive_traffic_test)
        set_tests_properties(alive_traffic PROPERTIES TIMEOUT 900)
    endif()
endif()
