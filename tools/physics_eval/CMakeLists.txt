# The physics backend evaluation prototype: an isolated project, NOT part of the engine build. It builds Jolt from a fetched source tree (tools/physics_eval/fetch_jolt.py) and links the system Bullet
# (apt libbullet-dev), and runs the same workloads through one backend interface.
#   python3 tools/physics_eval/fetch_jolt.py /tmp/physics_eval/jolt
#   cmake -S tools/physics_eval -B /tmp/physics_eval/build -DJOLT_ROOT=/tmp/physics_eval/jolt -DCMAKE_BUILD_TYPE=Release && cmake --build /tmp/physics_eval/build -j
cmake_minimum_required(VERSION 3.21)
project(saige_physics_eval LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
set(JOLT_ROOT "" CACHE PATH "Directory holding Jolt/ (from fetch_jolt.py)")
if(NOT EXISTS "${JOLT_ROOT}/Jolt/Jolt.h")
    message(FATAL_ERROR "JOLT_ROOT must hold Jolt/Jolt.h: run tools/physics_eval/fetch_jolt.py DIR and pass -DJOLT_ROOT=DIR")
endif()
file(GLOB_RECURSE JOLT_SOURCES "${JOLT_ROOT}/Jolt/*.cpp")
add_library(jolt STATIC ${JOLT_SOURCES})
target_include_directories(jolt PUBLIC "${JOLT_ROOT}")
option(JOLT_DOUBLE "Build Jolt with double-precision positions (JPH_DOUBLE_PRECISION)" OFF)
if(JOLT_DOUBLE)
    target_compile_definitions(jolt PUBLIC JPH_DOUBLE_PRECISION)
endif()
target_compile_options(jolt PRIVATE -ffp-contract=off -w)
target_compile_definitions(jolt PUBLIC "$<$<CONFIG:Release>:NDEBUG>")
find_package(Bullet REQUIRED)
find_package(Threads REQUIRED)
add_executable(physics_eval eval.cpp jolt_backend.cpp bullet_backend.cpp)
target_include_directories(physics_eval PRIVATE ${BULLET_INCLUDE_DIRS})
target_link_libraries(physics_eval PRIVATE jolt ${BULLET_LIBRARIES} Threads::Threads)
target_compile_options(physics_eval PRIVATE -ffp-contract=off -Wall -Wextra)
